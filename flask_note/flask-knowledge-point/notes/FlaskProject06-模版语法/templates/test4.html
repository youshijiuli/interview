{# 当变量数据没有传入，可以给变量设置默认值 #}

{{ name | default('游客') }}

{# 字符串的大小写转换 #}

{{ 'XYZ' | lower }}

{# 反转字符串 #}

{{ 'hello' | reverse }}

{# 格式化字符串 #}

{{ '今天是%d号，天气%s' | format(17, '晴') }}

{# 安全模式，禁止HTML自动转义 #}

{{ 'name' | safe }}

{# 去掉前后的空白字符 #}

{{ ' hello world ' | trim }}

{# 四舍五入 #}

{{ 3.1415 | round }}

{# 保留n位小数 #}

{{ 3.1415 | round(3) }}

{# 取绝对值 #}

{{ -10 | abs }}

{# 保留n位小数 #}

{{ 3.1415 | round(3) }}

{# 定义一个numbers变量 #}

{% set numbers = [3, 4, 1, 2, 8, 5] %}

{# 获取列表的第一个元素 #}

{{ numbers | first }}

{# 获取列表的长度 #}

{{ numbers | length }}

{# 列表求和 #}

{{ numbers | sum }}

{# 列表排序 #}

{{ numbers | sort }}

{# 根据字典的内容来使用过滤器 #}

{{ student_list | sort(attribute='age') }}

    {% for stu in student_list | sort(attribute='age', reverse=false) %}
  • {{ stu['name'] }}, {{ stu.age }}
    • {% endfor %}
{# 给数据分组 #}
    {% for group in student_list | groupby('gender') %}
  • {{ group.grouper }}
    • {% for stu in group.list %}
  • {{ stu.name }}, {{stu.gender}}
    • {% endfor %} {% endfor %}
{# 取出字典中的某列，组成一个大的列表，然后再把列表变成字符串 #}

{{ student_list | map(attribute='name') }}

{{ student_list | map(attribute='name') | join(',') }}

{# 自定义的装饰使用 #}

{{ [2, 3, 1, 0, 8] | list_split }}

{{ [2, 3, 1, 0, 8] | get_even }}
