from django.shortcuts import render, HttpResponse

def depart_multi(request):
    """ 批量删除（Excel文件）"""
    from openpyxl import load_workbook

    # 1.获取用户上传的文件对象
    file_object = request.FILES.get("exc")

    # 2.对象传递给openpyxl，由openpyxl读取文件的内容
    wb = load_workbook(file_object)
    sheet = wb.worksheets[0]

    # 3.循环获取每一行数据
    for row in sheet.iter_rows(min_row=2):
        text = row[0].value
        exists = models.Department.objects.filter(title=text).exists()
        if not exists:
            models.Department.objects.create(title=text)

    return redirect('/depart/list/')
```

from django import forms
from app01.utils.bootstrap import BootStrapForm


class UpForm(BootStrapForm):
    bootstrap_exclude_fields = ['img']

    name = forms.CharField(label="姓名")
    age = forms.IntegerField(label="年龄")
    img = forms.FileField(label="头像")


def upload_form(request):
    title = "Form上传"
    if request.method == "GET":
        form = UpForm()
        return render(request, 'upload_form.html', {"form": form, "title": title})

    form = UpForm(data=request.POST, files=request.FILES)
    if form.is_valid():
        # {'name': '武沛齐', 'age': 123, 'img': <InMemoryUploadedFile: 图片 1.png (image/png)>}
        # 1.读取图片内容，写入到文件夹中并获取文件的路径。
        image_object = form.cleaned_data.get("img")

        # file_path = "app01/static/img/{}".format(image_object.name)
        db_file_path = os.path.join("static", "img", image_object.name)

        file_path = os.path.join("app01", db_file_path)
        f = open(file_path, mode='wb')
        for chunk in image_object.chunks():
            f.write(chunk)
        f.close()

        # 2.将图片文件路径写入到数据库
        models.Boss.objects.create(
            name=form.cleaned_data['name'],
            age=form.cleaned_data['age'],
            img=db_file_path,
        )
        return HttpResponse("...")
    return render(request, 'upload_form.html', {"form": form, "title": title})

```



注意：就目前而言，所有的静态文件都只能放在static目录。



在django的开发过程中两个特殊的文件夹：

- static，存放静态文件的路径，包括：CSS、JS、项目图片。
- media，用户上传的数据的目录。

## 9.3. 上传文件

- 用户往往需要通过浏览器向网站提交各种各样的文件

### 19.3.1. 前端表单上传文件

- 规则：
	1. 文件上传必须为POST
	2. 表单 'form' 中文件上传必须带有 enctype='multipart/form-data' 才会包含有文件内容的数据
	3. 表单中用 \<input type="file" name="xxx"\> 标签上传文件

```html
<form action="/note/add_note_by_file" method="post" enctype="multipart/form-data">
    <p>标题：<input type="text" name="title"></p>
    <p><input type="file" name="file"></p>
    <p><input type="submit" value="提交"></p>
</form>
```

### 19.3.2. 后端接收上传的文件配置路径

- request.FILES.get(key) 获取文件对象
	1. key对应前端表单中File框的name值
	2. file绑定文件流对象
	3. file.name 文件名
	4. file.file 文件的字节流数据

1. 配置文件访问路径和 <kbd>存储路径</kbd>
	1. 在setting.py设置MEDIA相关配置
	2. Django把用户上传的文件统称为media资源

```python
# settings.py
MEDIA_URL = '/media/'
MEDIA_ROOT = os.path.join(BASE_DIR, 'media')
```

2. 创建 media （可以换名字）这个文件夹，在主路由中添加文件夹路由

### 19.3.3. 接收上传文件

1. 【不推荐】后端接收上传的文件并存储（以一个视图功能为例）
	- 为了确保编码问题，每一个str转换和打开文件都建议加上 <kbd>encoding='utf-8'</kbd> 
	- 这样子有问题： <font color=#ff00>重名、细节处理不够</font>

```python
@login_required
def add_note_by_file(request):
    # 1. 获取用户
    u = request.user
    # 2. 读取提交的数据
    # 2.1. 题目
    title = request.POST.get('title', '')
    # 2.2. 文件内容
    file = request.FILES.get('file', '')
    content = str(file.file.getvalue(), encoding='utf-8')
    # 3. 存储成文件
    filename = os.path.join(settings.MEDIA_ROOT, str(title)+'.txt')
    if not os.path.exists(filename):
        with open(filename, 'w+', encoding='utf-8') as f:
            f.write(content)
    # 4. 存储笔记数据到数据库
    Note.objects.create(title=title, content=content, user_info=u)

    return HttpResponseRedirect('/note/list')
```

2. 借助ORM接收文件（FileField对象）
	- 首先在models中添加FileField字段
	- 使用这个字段，调用create方法会自动保存文件
	- 保存文件时会**处理同名文件**

```python
# models.py
class Note(models.Model):
    ... # 其他字段

    # upload_to 指明子目录的名字
    # 比如指定存放根目录为files，给 upload_to 参数 notes
    # 那么最后文件存放在files/notes/ 文件夹中
    file = models.FileField(upload_to='notes', default='')

# views.py
@login_required
def add_note_by_file(request):
    # 1. 获取用户
    u = request.user
    # 2. 读取提交的数据
    # 2.1. 题目
    title = request.POST.get('title', '')
    # 2.2. 文件内容
    file = request.FILES.get('file', '')
    content = str(file.file.getvalue(), encoding='utf-8')
    # 3. 存储笔记数据到数据库（顺便帮忙存成文件了）
    Note.objects.create(title=title, content=content, file=file, user_info=u)

    return HttpResponseRedirect('/note/list')
```

## 19.4. 响应下载文件请求

- 设置标头符合文件传输规则
	- 设定 response['Content_type'] = 'application/octet-stream' 为字节流传输
	- response['Content-Disposition'] = 'attachment; filename="文件名"') 指定传输文件文件名

1. 如果是【不推荐】方式上传的文件
	- 获取标题和内容
	- 如果有这个文件，那就返回，没有，创建后返回

```python
def download_note(request):
    # 1. 获取笔记
    note_id = int(request.GET.get("nid", '1'))
    try:
        note = Note.objects.get(id=note_id)
    except Exception:
        return HttpResponse(
            JsonBuilder.buildJson(code=JsonBuilder.FAILURE_CODE, info='fail to  download'))
    # 2. 获取内容
    title = note.title
    content = note.content
    # 3. 返回带文件的 response
    fileurl = os.path.join(os.path.join(os.getcwd(), "files"), str(title) + '.txt')
    if os.path.exists(fileurl):
        # 3.1.1. 如果有文件
        file = open(fileurl, 'r+', encoding='utf-8')
        file = file.read()
    else:
        # 3.1.2. 没有文件，文件生成
        file = open(fileurl, 'w+', encoding='utf-8')
        file.write(content)
        file = file.read()
    # 3.2. 文件存入标头
    response = HttpResponse(file)
    response['Content_type'] = 'application/octet-stream'
    # 中文文件名转码
    response['Content-Disposition'] = "attachment; filename*=utf-8''{}".format(escape_uri_path(title)+'.txt')

    return response
```

2. 如果使用【ORM方式】存储
	- 可以通过ORM字段获取文件名后进行操作，逻辑基本一致
	- 就是如果没有这个文件，那就创建后还有存入数据库（存的是文件路径，包括upload_to的路径）

```python
# 导入Django自带的File工具
from django.core.files import File

@login_required
def download_note(request):
    # 1. 获取笔记
    note_id = int(request.GET.get("nid", '1'))
    try:
        note = Note.objects.get(id=note_id)
    except Exception:
        return HttpResponse(
            JsonBuilder.buildJson(code=JsonBuilder.FAILURE_CODE, info='fail to  download'))
    # 2. 获取内容
    title = note.title
    content = note.content
    filename = note.file
    # 3. 返回带文件的 response
    if not filename:
        # 3.1.1. 没有文件，文件生成
        filename = os.path.join('notes', str(title) + '.txt')
        fileurl = os.path.join(os.getcwd(), os.path.join('files', filename))
        file = open(fileurl, 'w+', encoding='utf-8')
        file.write(content)
        # 3.1.2. 将文件（文件名）存回数据库
        # 先打包成Django的File对象
        file = File(file)
        # save=True会自动调用model的Save方法
        note.file.save(str(title) + '.txt', file, save=True)1
        # 删除临时文件
        os.remove(fileurl)

    # 3.2. 获取 FileField 对象，通过它获取文件
    file = File(note.file)
    file = file.read()

    # 3.2. 文件存入标头
    response = HttpResponse(file)
    response['Content_type'] = 'application/octet-stream'
    # 中文文件名转码
    response['Content-Disposition'] = "attachment; filename*=utf-8''{}".format(escape_uri_path(title)+'.txt')

    return response
```
